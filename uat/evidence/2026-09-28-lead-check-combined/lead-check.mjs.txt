import { chromium, request } from 'playwright'
const BASE = 'http://localhost:8931', CTRL = 'http://127.0.0.1:2180'
const E = process.env.EVID
const log = (...a) => console.log(...a)
const api = await request.newContext({ baseURL: BASE })
let r = await api.get('/health');
r = await api.post('/api/v1/onboarding/complete', { data: { provider: { auth_method: 'api_key', id: 'openrouter', api_key: process.env.OPENROUTER_API_KEY, model: 'z-ai/glm-5.3-flash' }, admin: { username: 'admin', password: 'admin1234' } } })
log('onboard', r.status())
r = await api.post('/api/v1/auth/login', { data: { username: 'admin', password: 'admin1234' } })
log('login', r.status())
const st = await api.storageState()
const csrf = st.cookies.find((c) => c.name === '__Host-csrf' || c.name === 'csrf').value
const H = { 'X-Csrf-Token': csrf }
r = await api.post('/api/v1/workspaces', { headers: H, data: { name: 'Lead check', core_team: ['mia'] } })
const ws = (await r.json()).id; log('workspace', r.status(), ws)
r = await api.put(`/api/v1/agents/mia/mailboxes/${ws}`, { headers: H, data: { enabled: true, imap_host: '127.0.0.1', imap_port: 2143, smtp_host: '127.0.0.1', smtp_port: 2025, username: 'agent@test.local', password: 's3cret', signature_html: '<p>Kind regards, Mia</p>' } })
log('mailbox', r.status(), (await r.text()).slice(0, 120))
r = await api.post(`${CTRL}/inject`, { data: { user: 'agent@test.local', subject: 'Welcome to review', text: 'Hello Mia, this is an inbound test message.' } })
log('inject', r.status(), await r.text())

const browser = await chromium.launch({ headless: true })
const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 }, storageState: st })
const page = await ctx.newPage()
const errs = []
page.on('pageerror', (e) => errs.push('pageerror ' + e.message))
page.on('console', (m) => { if (m.type() === 'error') errs.push(m.text()) })
page.on('response', (res) => { if (res.url().includes('/mail') && res.status() >= 400) errs.push(`HTTP ${res.status()} ${res.url()}`) })
const shot = (p, n) => p.screenshot({ path: `${E}/${n}.png` })
await page.goto(`${BASE}/#/workspaces/${ws}/chat?panel=mail&agent=mia`)
await page.waitForSelector('[data-testid="mail-panel"]', { timeout: 30000 })
await page.getByText('Welcome to review').first().waitFor({ timeout: 30000 })
const selText = await page.getByRole('combobox', { name: 'Mailbox' }).textContent()
log('R2 selector label:', JSON.stringify(selText))
await shot(page, '01-panel-list-stacked')
await page.getByText('Welcome to review').first().click()
await page.getByText('this is an inbound test message').first().waitFor({ timeout: 30000 })
const list = await page.getByText('Welcome to review').first().boundingBox()
const body = await page.getByText('this is an inbound test message').first().boundingBox()
log('R8 open OK; R9 stacked (preview below list):', body.y > list.y + 20, Math.round(list.y), Math.round(body.y))
await shot(page, '02-panel-message-open-stacked')
const popP = ctx.waitForEvent('page', { timeout: 20000 })
await page.click('[data-testid="panel-expand"]')
const pop = await popP
await pop.waitForLoadState('load')
await pop.getByText('this is an inbound test message').first().waitFor({ timeout: 30000 })
const pl = await pop.getByText('Welcome to review').first().boundingBox()
const pb = await pop.getByText('this is an inbound test message').first().boundingBox()
const chatVisible = await pop.getByPlaceholder(/Message /).count()
log('R10 expand url:', pop.url(), 'side-by-side:', pb.x > pl.x + 100, 'chat composer count:', chatVisible)
await pop.screenshot({ path: `${E}/03-fullpage-side-by-side.png` })
await pop.close()
await page.goto(`${BASE}/#/workspaces/${ws}/chat?panel=mail&agent=mia`)
await page.waitForSelector('[data-testid="mail-panel"]', { timeout: 30000 })
const unreadAfter = await page.locator('text=INBOX').locator('..').textContent()
log('B-27 inbox row after opening message:', JSON.stringify(unreadAfter))
await page.getByRole('button', { name: /Compose/ }).first().click()
await page.getByText('Compose message').first().waitFor({ timeout: 20000 })
const to = page.getByRole('combobox', { name: /^To/ }).or(page.getByRole('textbox', { name: /^To/ })).first()
await to.fill('alice@test.local'); await to.press('Enter')
await page.getByRole('textbox', { name: /Subject/ }).first().fill('Lead check send')
const ed = page.locator('[contenteditable="true"]').first()
await ed.click(); await page.keyboard.type('Hello Alice, ')
await page.keyboard.press('Meta+b'); await page.keyboard.type('bold part'); await page.keyboard.press('Meta+b')
await shot(page, '04-compose-editor')
await page.getByRole('button', { name: /^Send/ }).last().click()
await page.waitForTimeout(4000)
await shot(page, '05-after-send')
r = await api.get(`${CTRL}/smtp`)
const sm = await r.json()
const last = sm.messages?.[sm.messages.length - 1] || ''
log('smtp count', sm.count, 'has bold <strong>:', /<strong>bold part<\/strong>|<b>bold part<\/b>/.test(last), 'has signature:', last.includes('Kind regards, Mia'), 'to alice:', last.includes('alice@test.local'))
log('ERRORS', errs.length); errs.slice(0, 10).forEach((e) => log('  ', e.slice(0, 200)))
await browser.close()
